# Pi 5 (BCM2712). The bootloader is in the board's EEPROM, so this file plus
# the kernel and the dtb is the whole boot partition. A non-empty config.txt
# is what marks the partition bootable here, so do not empty it out.

# No fan, so cap the ceiling (stock 2400) and let the governor idle down.
arm_freq=1500

# Serial console on the 40-pin header.
enable_uart=1
dtoverlay=dwc2
kernel=kernel8.img
device_tree=bcm2712-rpi-5-b.dtb
dtoverlay=vc4-kms-v3d,noaudio
